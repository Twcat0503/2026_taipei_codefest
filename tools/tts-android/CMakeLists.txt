function(require_revision directory revision)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD
        OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
    if(NOT result EQUAL 0 OR NOT actual STREQUAL revision)
        message(FATAL_ERROR "Unexpected runtime revision in ${directory}: ${actual}")
    endif()
endfunction()

function(apply_patch directory patch)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" apply --reverse --check "${patch}"
        RESULT_VARIABLE reverse OUTPUT_QUIET ERROR_QUIET)
    if(NOT reverse EQUAL 0)
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" apply --check "${patch}"
            RESULT_VARIABLE check)
        if(NOT check EQUAL 0)
            message(FATAL_ERROR "Patch does not match pinned runtime: ${patch}")
        endif()
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" apply "${patch}" RESULT_VARIABLE result)
        if(NOT result EQUAL 0)
            message(FATAL_ERROR "Cannot apply ${patch}")
        endif()
    endif()
endfunction()

        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -I -X utf8
            "${CMAKE_CURRENT_SOURCE_DIR}/generate_bluemagpie_unicode.py" "${UNICODE_TABLE}" RESULT_VARIABLE result)
        if(NOT result EQUAL 0)
            message(FATAL_ERROR "Unicode generation requires tokenizers==0.23.3")
        endif()
    endif()
    file(SHA256 "${UNICODE_TABLE}" unicode_hash)
    if(NOT unicode_hash STREQUAL "668a2d27d7a27ef2f46a6769100a35379c52aee4f4a0f0e902b7f78dbc69340d")
        message(FATAL_ERROR "Unexpected Unicode table: do not build with an approximate tokenizer")

    add_executable(townpass-tts-server native/main.cpp "${VENDOR}/cpp-httplib/httplib.cpp")
    target_compile_definitions(townpass-tts-server PRIVATE TOWNPASS_TTS_RECEIVE_TIMEOUT_MS=5000)
    target_include_directories(townpass-tts-server PRIVATE "${VENDOR}" "${VENDOR}/cpp-httplib" ${TTS_BACKBONE_INCLUDE_DIRS})
